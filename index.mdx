export const HomeMotion = () => {
  const ref = useRef(null);

  useEffect(() => {
    const node = ref.current;
    const root = node && node.closest ? node.closest(".zl-home") : null;
    if (!root || typeof window.matchMedia !== "function") return undefined;

    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)");
    const fine = window.matchMedia("(hover: hover) and (pointer: fine)");
    const cleanups = [];

    // Scroll reveal: only blocks that start below the fold are hidden, once.
    if (!reduce.matches && "IntersectionObserver" in window) {
      const fold = window.innerHeight || document.documentElement.clientHeight;
      const pending = Array.from(root.querySelectorAll(".zl-reveal")).filter(
        (el) => el.getBoundingClientRect().top > fold
      );
      if (pending.length) {
        const timers = [];
        const io = new IntersectionObserver(
          (entries) => {
            let step = 0;
            entries.forEach((entry) => {
              if (!entry.isIntersecting) return;
              const el = entry.target;
              io.unobserve(el);
              el.style.transitionDelay = Math.min(step, 4) * 70 + "ms";
              step += 1;
              el.classList.remove("zl-pending");
              timers.push(window.setTimeout(() => { el.style.transitionDelay = ""; }, 1100));
            });
          },
          { rootMargin: "0px 0px -48px 0px", threshold: 0 }
        );
        pending.forEach((el) => {
          el.classList.add("zl-pending");
          io.observe(el);
        });
        cleanups.push(() => {
          io.disconnect();
          timers.forEach((t) => window.clearTimeout(t));
          pending.forEach((el) => {
            el.classList.remove("zl-pending");
            el.style.transitionDelay = "";
          });
        });
      }
    }

    // Ambient glow: follows a mouse or pen only, never touch.
    if (fine.matches) {
      const hosts = Array.from(root.querySelectorAll(".zl-glow-host"))
        .map((el) => ({
          el,
          layer: el.querySelector(".zl-glow"),
          blob: el.querySelector(".zl-glow-blob"),
          active: false,
          x: 0,
          y: 0,
          restX: 0,
          restY: 0,
        }))
        .filter((h) => h.layer && h.blob);
      let frame = 0;
      const place = (h) => {
        const r = h.layer.getBoundingClientRect();
        h.el.style.setProperty("--zl-dx", (h.x - r.left - h.restX).toFixed(1) + "px");
        h.el.style.setProperty("--zl-dy", (h.y - r.top - h.restY).toFixed(1) + "px");
      };
      const render = () => {
        frame = 0;
        hosts.forEach((h) => {
          if (h.active) place(h);
        });
      };
      const schedule = () => {
        if (!frame) frame = window.requestAnimationFrame(render);
      };
      const onScroll = () => {
        if (hosts.some((h) => h.active)) schedule();
      };

      hosts.forEach((h) => {
        const start = (e) => {
          h.restX = h.blob.offsetLeft;
          h.restY = h.blob.offsetTop;
          h.x = e.clientX;
          h.y = e.clientY;
          h.active = true;
          if (h.layer.classList.contains("zl-glow-soft")) {
            h.blob.style.transition = "none";
            place(h);
            h.blob.getBoundingClientRect();
            h.blob.style.transition = "";
          }
          h.el.classList.add("zl-glow-active");
        };
        const onMove = (e) => {
          if (e.pointerType === "touch" || reduce.matches) return;
          if (!h.active) start(e);
          h.x = e.clientX;
          h.y = e.clientY;
          schedule();
        };
        const onLeave = () => {
          h.active = false;
          h.el.classList.remove("zl-glow-active");
          h.el.style.removeProperty("--zl-dx");
          h.el.style.removeProperty("--zl-dy");
        };
        h.el.addEventListener("pointermove", onMove, { passive: true });
        h.el.addEventListener("pointerleave", onLeave);
        cleanups.push(() => {
          h.el.removeEventListener("pointermove", onMove);
          h.el.removeEventListener("pointerleave", onLeave);
          onLeave();
        });
      });

      window.addEventListener("scroll", onScroll, { passive: true, capture: true });
      cleanups.push(() => {
        window.removeEventListener("scroll", onScroll, { capture: true });
        if (frame) window.cancelAnimationFrame(frame);
      });
    }

    return () => cleanups.forEach((fn) => fn());
  }, []);

  return <span ref={ref} hidden />;
};
